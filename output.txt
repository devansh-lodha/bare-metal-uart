#![no_std]
#![no_main]

use core::arch::global_asm;
use core::panic::PanicInfo;
use aarch64_cpu::registers::*;
use tock_registers::interfaces::Readable;

mod uart;
mod mmu;

global_asm!(include_str!("start.s"));

#[no_mangle]
pub extern "C" fn rust_main() -> ! {
    unsafe {
        uart::init();
        uart::puts("\r\n\r\n");
        uart::puts("========================================\r\n");
        uart::puts("   RASPBERRY PI 5 - BARE METAL MMU      \r\n");
        uart::puts("========================================\r\n");

        let el = CurrentEL.read(CurrentEL::EL);
        uart::puts("CurrentEL: ");
        uart::print_hex(el);
        uart::puts("\r\n");

        uart::puts("Initializing MMU... ");
        mmu::init();
        uart::puts("Done.\r\n");

        let sctlr = SCTLR_EL1.get();
        uart::puts("SCTLR_EL1: ");
        uart::print_hex(sctlr);
        
        if (sctlr & 1) == 1 {
            uart::puts(" (MMU: ON)\r\n");
        } else {
            uart::puts(" (MMU: OFF)\r\n");
        }

        let ttbr = TTBR0_EL1.get_baddr();
        uart::puts("TTBR0_EL1: ");
        uart::print_hex(ttbr);
        uart::puts("\r\n");
        
        uart::puts("Entering Echo Loop...\r\n");
    }

    loop {
        let c = uart::getc();
        if c == '\r' {
            uart::puts("\r\n");
        } else {
            uart::putc(c);
        }
    }
}

#[panic_handler]
fn panic(_info: &PanicInfo) -> ! {
    uart::puts("\r\nPANIC\r\n");
    loop {}
}

#![allow(static_mut_refs)]

use core::ptr::addr_of_mut;
use aarch64_cpu::{asm::barrier, registers::*};
use tock_registers::{
    interfaces::{ReadWriteable, Writeable},
    register_bitfields,
    registers::InMemoryRegister,
};

// -----------------------------------------------------------------------------
// Constants
// -----------------------------------------------------------------------------
const BLOCK_SIZE: u64 = 512 * 1024 * 1024; // 512 MB

// MAIR Attributes
const MAIR_DEVICE: u64 = 0;
const MAIR_NORMAL: u64 = 1;

// -----------------------------------------------------------------------------
// Descriptor Bitfields (The Pro Way)
// -----------------------------------------------------------------------------
// This defines the layout of a Level 2 Block Descriptor as per ARM ARM
register_bitfields! {
    u64,

    // Stage 1 Block Descriptor
    STAGE1_DESCRIPTOR [
        // Execute Never (Privileged)
        PXN      OFFSET(53) NUMBITS(1) [],
        
        // Output Address (48-bit physical address)
        OUTPUT_ADDR_64KiB OFFSET(16) NUMBITS(32) [],

        // Access Flag (Must be 1 to avoid fault)
        AF       OFFSET(10) NUMBITS(1) [
            False = 0,
            True = 1
        ],

        // Shareability
        SH       OFFSET(8) NUMBITS(2) [
            OuterShareable = 0b10,
            InnerShareable = 0b11
        ],

        // Access Permissions
        AP       OFFSET(6) NUMBITS(2) [
            RW_EL1 = 0b00,
            RW_EL1_EL0 = 0b01,
            RO_EL1 = 0b10,
            RO_EL1_EL0 = 0b11
        ],

        // Memory Attribute Index (Matches MAIR)
        AttrIndx OFFSET(2) NUMBITS(3) [],

        // Descriptor Type (Block vs Table)
        TYPE     OFFSET(0) NUMBITS(2) [
            Block = 0b01, // Valid Block
            Table = 0b11, // Valid Table (Level 3 only)
            Invalid = 0b00
        ]
    ]
}

// -----------------------------------------------------------------------------
// Data Structures
// -----------------------------------------------------------------------------

#[repr(C)]
#[repr(align(65536))]
struct TranslationTable {
    entries: [u64; 8192],
}

static mut KERNEL_TABLE: TranslationTable = TranslationTable {
    entries: [0; 8192],
};

// -----------------------------------------------------------------------------
// Helpers
// -----------------------------------------------------------------------------

/// Create a Block Entry (Level 2) using safe, typed bitfields
fn make_block_desc(phys_addr: u64, mair_idx: u64, xn: bool) -> u64 {
    // 1. Create a temporary register wrapper to hold the value
    let val = InMemoryRegister::<u64, STAGE1_DESCRIPTOR::Register>::new(0);

    // 2. Set bits using named fields
    val.write(
        STAGE1_DESCRIPTOR::OUTPUT_ADDR_64KiB.val(phys_addr >> 16) +
        STAGE1_DESCRIPTOR::AF::True +
        STAGE1_DESCRIPTOR::SH::InnerShareable +
        STAGE1_DESCRIPTOR::AP::RW_EL1 +
        STAGE1_DESCRIPTOR::AttrIndx.val(mair_idx) +
        STAGE1_DESCRIPTOR::TYPE::Block +
        if xn { STAGE1_DESCRIPTOR::PXN.val(1) } else { STAGE1_DESCRIPTOR::PXN.val(0) }
    );

    val.get()
}

// -----------------------------------------------------------------------------
// Initialization
// -----------------------------------------------------------------------------

pub unsafe fn init() {
    // 1. Setup MAIR (Memory Attributes)
    MAIR_EL1.write(
        MAIR_EL1::Attr0_Device::nonGathering_nonReordering_EarlyWriteAck +
        MAIR_EL1::Attr1_Normal_Outer::WriteBack_NonTransient_ReadWriteAlloc +
        MAIR_EL1::Attr1_Normal_Inner::WriteBack_NonTransient_ReadWriteAlloc
    );

    // 2. Setup TCR (Translation Control)
    TCR_EL1.write(
        TCR_EL1::TBI0::Used +
        TCR_EL1::IPS::Bits_48 +
        TCR_EL1::TG0::KiB_64 +
        TCR_EL1::SH0::Inner +
        TCR_EL1::ORGN0::WriteBack_ReadAlloc_WriteAlloc_Cacheable +
        TCR_EL1::IRGN0::WriteBack_ReadAlloc_WriteAlloc_Cacheable +
        TCR_EL1::EPD1::DisableTTBR1Walks + 
        TCR_EL1::T0SZ.val(64 - 42) // 42 bits VA
    );

    // 3. Populate Page Table
    let table_ptr = addr_of_mut!(KERNEL_TABLE);
    
    // A. Map RAM (0x0) -> Normal, Executable
    (*table_ptr).entries[0] = make_block_desc(0x0, MAIR_NORMAL, false);

    // B. Map IO (0x1f_0000_0000) -> Device, Execute Never
    let io_base = 0x1f_0000_0000;
    let io_idx = (io_base / BLOCK_SIZE) as usize;
    
    // Map 2 blocks (1GB) for peripherals
    (*table_ptr).entries[io_idx] = make_block_desc(io_base, MAIR_DEVICE, true);
    (*table_ptr).entries[io_idx+1] = make_block_desc(io_base + BLOCK_SIZE, MAIR_DEVICE, true);

    // 4. Set TTBR0
    TTBR0_EL1.set_baddr(table_ptr as u64);

    // 5. Enable MMU and Caches
    barrier::isb(barrier::SY);

    SCTLR_EL1.modify(
        SCTLR_EL1::M::Enable +  
        SCTLR_EL1::C::Cacheable + 
        SCTLR_EL1::I::Cacheable   
    );

    barrier::isb(barrier::SY);
}

.section .text.boot
.global _start

_start:
    // 1. Check Core ID (Only Core 0)
    mrs     x0, mpidr_el1
    and     x0, x0, #3
    cbz     x0, .L_check_el

.L_park:
    wfe
    b       .L_park

.L_check_el:
    // 2. Check current Exception Level
    mrs     x0, CurrentEL
    and     x0, x0, #0b1100
    lsr     x0, x0, #2
    
    // If EL2, prepare transition. If EL1, go to stack setup.
    cmp     x0, #2
    beq     .L_el2_entry
    b       .L_setup_stack

.L_el2_entry:
    // ----------------------------------------------------
    // CRITICAL RPi5 EL2 -> EL1 SETUP
    // ----------------------------------------------------

    // A. Enable Timer access for EL1
    mov     x0, #0b11       // EL1PCEN | EL1PCTEN
    msr     cnthctl_el2, x0
    msr     cntvoff_el2, xzr

    // B. Set HCR_EL2 (Hypervisor Config) to AArch64
    mov     x0, #(1 << 31)  // RW bit
    msr     hcr_el2, x0

    // C. Setup SCTLR_EL1 (Safe Defaults: No MMU, No Cache)
    // 0x30C50830 is a safe reset value for Cortex-A76
    ldr     x0, =0x30C50830
    msr     sctlr_el1, x0

    // D. Enable FP/SIMD at EL1 (CPACR_EL1) - CRITICAL FOR RUST
    // Bits 20:21 (FPEN) = 11 (Don't trap FP instructions)
    mov     x0, #(3 << 20)
    msr     cpacr_el1, x0

    // E. Setup SPSR_EL2 (Return to EL1h, Mask Interrupts)
    // DAIF masked (1111) | EL1h (0101) = 0x3c5
    mov     x0, #0x3c5
    msr     spsr_el2, x0

    // F. Setup Return Address
    adr     x0, .L_setup_stack
    msr     elr_el2, x0

    // G. Invalidate I-Cache before dropping
    ic      iallu
    dsb     nsh
    isb

    // H. Drop to EL1
    eret

.L_setup_stack:
    // 3. Setup Stack at 0x80000 (Growing Down)
    ldr     x0, =_start
    mov     sp, x0

    // 4. Zero BSS (Mandatory for Rust statics)
    ldr     x0, =__bss_start
    ldr     x1, =__bss_end
    sub     x1, x1, x0
    cbz     x1, .L_main

.L_bss_loop:
    str     xzr, [x0], #8
    sub     x1, x1, #8
    cbnz    x1, .L_bss_loop

.L_main:
    bl      rust_main

.L_halt:
    wfe
    b       .L_halt

use core::ptr::{read_volatile, write_volatile};

// RP1 / Pi 5 Constants
const UART0_BASE: u64 = 0x1f00030000;
const UART_DR:   u64 = 0x00;
const UART_FR:   u64 = 0x18;
const UART_IBRD: u64 = 0x24;
const UART_FBRD: u64 = 0x28;
const UART_LCRH: u64 = 0x2c;
const UART_CR:   u64 = 0x30;
const UART_ICR:  u64 = 0x44;

const UART_FR_TXFF: u32 = 1 << 5;
const UART_FR_RXFE: u32 = 1 << 4;
const UART_LCRH_WLEN8: u32 = 0b11 << 5;
const UART_LCRH_FEN:   u32 = 1 << 4;
const UART_CR_UARTEN: u32 = 1 << 0;
const UART_CR_TXE:    u32 = 1 << 8;
const UART_CR_RXE:    u32 = 1 << 9;

const RP1_PERIPH_BASE: u64 = 0x1f00000000;
const IO_BANK0_BASE: u64 = RP1_PERIPH_BASE + 0xd0000;
const PADS_BANK0_BASE: u64 = RP1_PERIPH_BASE + 0xf0000;
const GPIO14_CTRL: u64 = IO_BANK0_BASE + 0x074;
const PADS14_CTRL: u64 = PADS_BANK0_BASE + 0x03c;
const GPIO15_CTRL: u64 = IO_BANK0_BASE + 0x07c;
const PADS15_CTRL: u64 = PADS_BANK0_BASE + 0x040;
const BIT_PADS_OD: u32 = 1 << 7;
const BIT_PADS_IE: u32 = 1 << 6;
const BIT_PADS_PUE: u32 = 1 << 3;
const FUNCSEL_MASK: u32 = 0x1F;
const FUNC_UART0: u32 = 4;

pub unsafe fn init() {
    let mut val = read_volatile(PADS14_CTRL as *mut u32);
    val &= !BIT_PADS_OD;
    write_volatile(PADS14_CTRL as *mut u32, val);
    
    let mut val = read_volatile(GPIO14_CTRL as *mut u32);
    val &= !FUNCSEL_MASK;
    val |= FUNC_UART0;
    write_volatile(GPIO14_CTRL as *mut u32, val);

    let mut val = read_volatile(PADS15_CTRL as *mut u32);
    val |= BIT_PADS_IE | BIT_PADS_PUE;
    write_volatile(PADS15_CTRL as *mut u32, val);

    let mut val = read_volatile(GPIO15_CTRL as *mut u32);
    val &= !FUNCSEL_MASK;
    val |= FUNC_UART0;
    write_volatile(GPIO15_CTRL as *mut u32, val);

    write_volatile((UART0_BASE + UART_CR) as *mut u32, 0);
    write_volatile((UART0_BASE + UART_IBRD) as *mut u32, 26);
    write_volatile((UART0_BASE + UART_FBRD) as *mut u32, 3);
    write_volatile((UART0_BASE + UART_LCRH) as *mut u32, UART_LCRH_FEN | UART_LCRH_WLEN8);
    write_volatile((UART0_BASE + UART_ICR) as *mut u32, 0x7FF);
    write_volatile((UART0_BASE + UART_CR) as *mut u32, UART_CR_UARTEN | UART_CR_TXE | UART_CR_RXE);
}

pub fn putc(c: char) {
    unsafe {
        while read_volatile((UART0_BASE + UART_FR) as *mut u32) & UART_FR_TXFF != 0 {}
        write_volatile((UART0_BASE + UART_DR) as *mut u32, c as u32);
    }
}

pub fn puts(s: &str) {
    for c in s.chars() {
        putc(c);
    }
}

pub fn getc() -> char {
    unsafe {
        while read_volatile((UART0_BASE + UART_FR) as *mut u32) & UART_FR_RXFE != 0 {}
        (read_volatile((UART0_BASE + UART_DR) as *mut u32) & 0xFF) as u8 as char
    }
}

// NEW: Helper to print hex values (0x1234...)
pub fn print_hex(mut val: u64) {
    puts("0x");
    
    // If 0, handle explicitly
    if val == 0 {
        puts("0");
        return;
    }

    // Determine number of digits (max 16)
    // We push to a temp buffer (stack) to reverse order
    let mut buffer = [0u8; 16];
    let mut idx = 0;

    while val > 0 {
        let digit = (val % 16) as u8;
        if digit < 10 {
            buffer[idx] = digit + b'0';
        } else {
            buffer[idx] = digit - 10 + b'a';
        }
        val /= 16;
        idx += 1;
    }

    // Print in reverse
    while idx > 0 {
        idx -= 1;
        putc(buffer[idx] as char);
    }
}
